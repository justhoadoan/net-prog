\begin{frame}{2. Proposed features and learning workflow}

  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{topicblock}{01\quad Study at the right level}
        \small
        \begin{itemize}
          \setlength{\itemsep}{0.2em}
          \setlength{\parsep}{0pt}
          \item Browse lessons by topic and proficiency level.
          \item Watch videos, listen to audio and read texts with vocabulary and grammar.
        \end{itemize}
      \end{topicblock}

        \small
        \begin{itemize}
          \setlength{\itemsep}{0.2em}
          \setlength{\parsep}{0pt}
          \item \textbf{Quizzes:} multiple choice, gap filling and sentence ordering.
          \item \textbf{Exercises:} sentence rewriting, paragraph writing and recorded speaking.
          \item Automatic scores for objective tasks; teacher feedback for writing and speaking.

    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{topicblock}{03\quad Play and communicate}

          \item \textbf{Games:} word/picture matching, sentence ordering and image puzzles.
          \item Private chat and real-time voice calls with other learners or teachers.
        \end{itemize}
      \end{topicblock}

      \begin{topicblock}{04\quad Create and manage content}
        \small
        \begin{itemize}

  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{topicblock}{Our application protocol: ELP/1}
        \small
        \begin{itemize}
          \setlength{\itemsep}{0.22em}
          \setlength{\parsep}{0pt}
          \item Length-prefixed JSON messages over TCP
          \item Requests, responses and server-pushed events
          \item Team-defined commands and session states
          \item UDP voice packets carry sequence numbers and audio samples
        \end{itemize}
      \end{topicblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{topicblock}{Socket handling and reliability}
        \small
        \begin{itemize}
          \setlength{\itemsep}{0.22em}
          \setlength{\parsep}{0pt}
          \item Standard JDK socket APIs
          \item Multiple clients, partial reads/writes, timeouts and disconnects
          \item Server logs every sent and received message
        \end{itemize}
      \end{topicblock}
    \end{column}
  \end{columns}
\end{frame}
